**db/transactions.ts:**
```typescript
import { ClientSession, ObjectId } from 'mongodb';
import { getClient } from './connection.js';
import { getOperationsCollection } from './collections.js';
import type { OperationType, IOperation } from '@solve/common';

/**
 * Execute a function within a MongoDB transaction.
 * Automatically handles session lifecycle and rollback on error.
 *
 * @param fn - Callback receiving session for transactional operations
 * @returns Result of the callback function
 * @throws Re-throws any error from callback after rollback
 */
export async function withTransaction<T>(
  fn: (session: ClientSession) => Promise<T>
): Promise<T> {
  const client = getClient();
  if (!client) {
    throw new Error('Database not connected');
  }

  const session = client.startSession();

  try {
    let result: T;
    await session.withTransaction(async () => {
      result = await fn(session);
    });
    return result!;
  } finally {
    await session.endSession();
  }
}

/**
 * Create an operation log entry within a transaction.
 * Should be called within withTransaction callback.
 *
 * @param session - MongoDB session from withTransaction
 * @param operation - Operation details (without _id and timestamp)
 * @returns Inserted operation with _id and timestamp
 */
export async function createOperationLog(
  session: ClientSession,
  operation: Omit<IOperation, '_id' | 'timestamp'>
): Promise<IOperation> {
  const operations = await getOperationsCollection();

  const doc: IOperation = {
    _id: new ObjectId().toString(),
    ...operation,
    timestamp: new Date(),
  };

  await operations.insertOne(doc, { session });
  return doc;
}
```

Key patterns per CONTEXT.md:
- Sync in transaction: mutation + operation log in same MongoDB transaction
- Use session.withTransaction() for automatic retry on transient errors
- Always endSession() in finally block to prevent leaks

```typescript
import { describe, it, expect, beforeAll, afterAll, beforeEach } from 'vitest';
import { ObjectId } from 'mongodb';
import { connectToDatabase, closeDatabase, getClient } from '../../src/db/connection.js';
import { getNodesCollection, getOperationsCollection } from '../../src/db/collections.js';
import { withTransaction, createOperationLog } from '../../src/db/transactions.js';
import type { INode } from '@solve/common';

describe('Transaction utilities', () => {
  beforeAll(async () => {
    await connectToDatabase();
  });

  afterAll(async () => {
    await closeDatabase();
  });

  beforeEach(async () => {
    // Clean up test data
    const nodes = await getNodesCollection();
    const operations = await getOperationsCollection();
    await nodes.deleteMany({ graphId: 'test-graph' });
    await operations.deleteMany({ graphId: 'test-graph' });
  });

  describe('withTransaction', () => {
    it('executes callback and returns result on success', async () => {
      const result = await withTransaction(async (session) => {
        return 'success';
      });
      expect(result).toBe('success');
    });

    it('commits both node and operation on success', async () => {
      const nodeId = new ObjectId().toString();

      await withTransaction(async (session) => {
        const nodes = await getNodesCollection();
        const node: INode = {
          _id: nodeId,
          graphId: 'test-graph',
          type: 'idea',
          label: 'Test Node',
          position: { x: 0, y: 0 },
          data: {},
          createdAt: new Date(),
          updatedAt: new Date(),
        };

        await nodes.insertOne(node, { session });

        await createOperationLog(session, {
          graphId: 'test-graph',
          userId: 'test-user',
          type: 'CREATE_NODE',
          target: { collection: 'nodes', id: nodeId },
          data: node,
        });
      });

      // Verify both committed
      const nodes = await getNodesCollection();
      const operations = await getOperationsCollection();

      const foundNode = await nodes.findOne({ _id: nodeId });
      const foundOp = await operations.findOne({
        graphId: 'test-graph',
        type: 'CREATE_NODE',
      });

      expect(foundNode).not.toBeNull();
      expect(foundOp).not.toBeNull();
      expect(foundOp?.target.id).toBe(nodeId);
    });

    it('rolls back all changes when callback throws after inserts', async () => {
      const nodeId = new ObjectId().toString();

      await expect(
        withTransaction(async (session) => {
          const nodes = await getNodesCollection();
          const node: INode = {
            _id: nodeId,
            graphId: 'test-graph',
            type: 'idea',
            label: 'Test Node',
            position: { x: 0, y: 0 },
            data: {},
            createdAt: new Date(),
            updatedAt: new Date(),
          };

          await nodes.insertOne(node, { session });

          await createOperationLog(session, {
            graphId: 'test-graph',
            userId: 'test-user',
            type: 'CREATE_NODE',
            target: { collection: 'nodes', id: nodeId },
            data: node,
          });

          // Simulate error after successful inserts
          throw new Error('Simulated failure');
        })
      ).rejects.toThrow('Simulated failure');

      // Verify both rolled back
      const nodes = await getNodesCollection();
      const operations = await getOperationsCollection();

      const foundNode = await nodes.findOne({ _id: nodeId });
      const foundOp = await operations.findOne({
        target: { id: nodeId },
      });

      expect(foundNode).toBeNull();
      expect(foundOp).toBeNull();
    });
  });

  describe('createOperationLog', () => {
    it('creates operation with correct fields', async () => {
      const nodeId = new ObjectId().toString();

      const operation = await withTransaction(async (session) => {
        return createOperationLog(session, {
          graphId: 'test-graph',
          userId: 'test-user',
          type: 'CREATE_NODE',
          target: { collection: 'nodes', id: nodeId },
          data: { label: 'Test' },
        });
      });

      expect(operation._id).toBeDefined();
      expect(operation.timestamp).toBeInstanceOf(Date);
      expect(operation.type).toBe('CREATE_NODE');
      expect(operation.target.collection).toBe('nodes');
    });
  });
});
```
