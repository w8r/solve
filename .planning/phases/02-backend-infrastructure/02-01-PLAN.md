**app.ts (exported builder for testing):**
```typescript
import Fastify, { FastifyInstance } from 'fastify';
import { getEnv } from './config/index.js';

interface BuildOptions {
  logger?: boolean | object;
}

export async function build(opts: BuildOptions = {}): Promise<FastifyInstance> {
  const env = getEnv();

  const server = Fastify({
    logger: opts.logger ?? {
      level: env.LOG_LEVEL,
      ...(env.NODE_ENV === 'development' && {
        transport: {
          target: 'pino-pretty',
          options: { colorize: true },
        },
      }),
    },
  });

  // Register plugins in order (infrastructure first, routes last)
  await server.register(import('./plugins/cors.js'));
  await server.register(import('./plugins/mongodb.js'));
  await server.register(import('./plugins/error-handler.js'));

  // Register routes
  await server.register(import('./routes/index.js'));

  return server;
}
```

```typescript
import type { FastifyPluginAsync } from 'fastify';
import { getClient } from '../../db/index.js';

const healthRoutes: FastifyPluginAsync = async (server) => {
  server.get('/health', async (request, reply) => {
    // Check database connectivity
    const client = getClient();
    let dbStatus = 'disconnected';
    if (client) {
      try {
        await client.db().admin().ping();
        dbStatus = 'connected';
      } catch {
        dbStatus = 'error';
      }
    }

    return {
      status: 'ok',
      timestamp: new Date().toISOString(),
      uptime: process.uptime(),
      database: dbStatus,
    };
  });
};

export default healthRoutes;
```
