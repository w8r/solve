```typescript
import type { INode, IEdge } from "../types/index.js";

type NodeVisitor = (node: INode) => void;

export function dfs(nodes: INode[], edges: IEdge[], startNodeId: string, visit: NodeVisitor): void {
  const adjacency = buildAdjacencyMap(nodes, edges);
  const visited = new Set<string>();

  function traverse(nodeId: string): void {
    if (visited.has(nodeId)) return;
    visited.add(nodeId);

    const node = nodes.find((n) => n._id === nodeId);
    if (node) visit(node);

    const neighbors = adjacency.get(nodeId) ?? [];
    for (const neighborId of neighbors) {
      traverse(neighborId);
    }
  }

  traverse(startNodeId);
}

export function bfs(nodes: INode[], edges: IEdge[], startNodeId: string, visit: NodeVisitor): void {
  const adjacency = buildAdjacencyMap(nodes, edges);
  const visited = new Set<string>();
  const queue: string[] = [startNodeId];

  while (queue.length > 0) {
    const nodeId = queue.shift()!;
    if (visited.has(nodeId)) continue;
    visited.add(nodeId);

    const node = nodes.find((n) => n._id === nodeId);
    if (node) visit(node);

    const neighbors = adjacency.get(nodeId) ?? [];
    for (const neighborId of neighbors) {
      if (!visited.has(neighborId)) {
        queue.push(neighborId);
      }
    }
  }
}

function buildAdjacencyMap(nodes: INode[], edges: IEdge[]): Map<string, string[]> {
  const map = new Map<string, string[]>();
  for (const node of nodes) {
    map.set(node._id, []);
  }
  for (const edge of edges) {
    const neighbors = map.get(edge.sourceId);
    if (neighbors) {
      neighbors.push(edge.targetId);
    }
  }
  return map;
}
```

```typescript
import type { IGraph, INode, IEdge } from "../types/index.js";

export interface ValidationResult {
  valid: boolean;
  errors: string[];
}

export function validateGraph(graph: IGraph, nodes: INode[], edges: IEdge[]): ValidationResult {
  const errors: string[] = [];
  const nodeIds = new Set(nodes.map((n) => n._id));

  // Check for orphan edges (edges pointing to non-existent nodes)
  for (const edge of edges) {
    if (!nodeIds.has(edge.sourceId)) {
      errors.push(`Edge ${edge._id} references non-existent source node ${edge.sourceId}`);
    }
    if (!nodeIds.has(edge.targetId)) {
      errors.push(`Edge ${edge._id} references non-existent target node ${edge.targetId}`);
    }
    if (edge.graphId !== graph._id) {
      errors.push(`Edge ${edge._id} has graphId ${edge.graphId} but expected ${graph._id}`);
    }
  }

  // Check node graph references
  for (const node of nodes) {
    if (node.graphId !== graph._id) {
      errors.push(`Node ${node._id} has graphId ${node.graphId} but expected ${graph._id}`);
    }
  }

  return {
    valid: errors.length === 0,
    errors,
  };
}
```

```typescript
import type { INode, IEdge } from "../types/index.js";

/**
 * Find all nodes that depend on the given node (direct and transitive)
 */
export function findDependents(nodeId: string, nodes: INode[], edges: IEdge[]): string[] {
  const dependents: string[] = [];
  const visited = new Set<string>();

  // Edges where sourceId is our node = nodes that depend on us
  function traverse(currentId: string): void {
    for (const edge of edges) {
      if (edge.sourceId === currentId && !visited.has(edge.targetId)) {
        visited.add(edge.targetId);
        dependents.push(edge.targetId);
        traverse(edge.targetId);
      }
    }
  }

  traverse(nodeId);
  return dependents;
}

/**
 * Find all nodes that the given node depends on (direct and transitive)
 */
export function findDependencies(nodeId: string, nodes: INode[], edges: IEdge[]): string[] {
  const dependencies: string[] = [];
  const visited = new Set<string>();

  function traverse(currentId: string): void {
    for (const edge of edges) {
      if (edge.targetId === currentId && !visited.has(edge.sourceId)) {
        visited.add(edge.sourceId);
        dependencies.push(edge.sourceId);
        traverse(edge.sourceId);
      }
    }
  }

  traverse(nodeId);
  return dependencies;
}

/**
 * Detect cycles in the graph
 */
export function hasCycle(nodes: INode[], edges: IEdge[]): boolean {
  const visiting = new Set<string>();
  const visited = new Set<string>();

  const adjacency = new Map<string, string[]>();
  for (const node of nodes) {
    adjacency.set(node._id, []);
  }
  for (const edge of edges) {
    adjacency.get(edge.sourceId)?.push(edge.targetId);
  }

  function dfs(nodeId: string): boolean {
    if (visiting.has(nodeId)) return true; // Cycle detected
    if (visited.has(nodeId)) return false;

    visiting.add(nodeId);
    for (const neighbor of adjacency.get(nodeId) ?? []) {
      if (dfs(neighbor)) return true;
    }
    visiting.delete(nodeId);
    visited.add(nodeId);
    return false;
  }

  for (const node of nodes) {
    if (dfs(node._id)) return true;
  }
  return false;
}
```
